export default async function handler(req, res) { if (req.method !== 'POST') { res.status(405).json({ error: { message: 'POSTのみ対応しています。' } }); return; } const apiKey = process.env.ANTHROPIC_API_KEY; if (!apiKey) { res.status(500).json({ error: { message: 'サーバーにANTHROPIC_API_KEYが設定されていません。' } }); return; } try { const { messages } = req.body; const response = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': apiKey, 'anthropic-version': '2023-06-01' }, body: JSON.stringify({ model: 'claude-sonnet-5', max_tokens: 1000, messages: messages }) }); const data = await response.json(); res.status(response.status).json(data); } catch (err) { res.status(500).json({ error: { message: 'サーバー内部でエラーが発生しました。' } }); } }
